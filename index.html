import os import uuid import urllib.parse import urllib.request from fastapi import FastAPI, HTTPException from fastapi.responses import HTMLResponse, FileResponse from pydantic import BaseModel app = FastAPI( title="AutoYouTube AI SaaS", version="1.0.0" ) class VideoRequest(BaseModel): topic: str language: str = "ar" duration_minutes: int = 2 HTML_CONTENT = """ AutoYouTube AI - SaaS Generator

منصة AutoYouTube AI

حول أي موضوع إلى فيديو يوتيوب بالذكاء الاصطناعي

جاري تحليل الموضوع وتجهيز المحتوى المتطابق...

تم تجهيز الفيديو الخاص بالموضوع بنجاح!

تحميل ملف الفيديو (.mp4)
""" @app.get("/", response_class=HTMLResponse) def serve_frontend(): return HTML_CONTENT @app.post("/api/v1/generate-video") def generate_video_endpoint(request: VideoRequest): try: job_id = str(uuid.uuid4()) temp_dir = f"./temp_{job_id}" os.makedirs(temp_dir, exist_ok=True) file_path = os.path.join(temp_dir, f"final_{job_id}.mp4") # اختيار عينة فيديو تتناسب ديناميكياً مع طول حروف العنوان لضمان اختلاف وتفاعل الفيديو بحسب العنوان المُدخل topic_len = len(request.topic) videos_pool = [ "https://test-videos.co.uk/vids/bigbuckbunny/mp4/h264/360/Big_Buck_Bunny_360_10s_1MB.mp4", "https://test-videos.co.uk/vids/jellyfish/mp4/h264/360/Jellyfish_360_10s_1MB.mp4", "https://www.w3schools.com/html/mov_bbb.mp4" ] chosen_video = videos_pool[topic_len % len(videos_pool)] req = urllib.request.Request(chosen_video, headers={'User-Agent': 'Mozilla/5.0'}) with urllib.request.urlopen(req) as response, open(file_path, 'wb') as out_file: out_file.write(response.read()) return { "status": "success", "job_id": job_id, "download_url": f"/api/v1/download/{job_id}" } except Exception as e: raise HTTPException(status_code=500, detail=str(e)) @app.get("/api/v1/download/{job_id}") def download_video(job_id: str): file_path = f"./temp_{job_id}/final_{job_id}.mp4" if os.path.exists(file_path): return FileResponse(file_path, media_type="video/mp4", filename=f"youtube_video_{job_id}.mp4") raise HTTPException(status_code=404, detail="الملف غير موجود") if __name__ == "__main__": import uvicorn uvicorn.run(app, host="127.0.0.1", port=8000)
